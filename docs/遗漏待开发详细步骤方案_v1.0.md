## 0.2 S-07-a · 平台接线守卫脚本

**新建**：`scripts/check_platform_wiring.py`

```python
#!/usr/bin/env python3
"""平台层组件接线守卫：扫描 lib/platform/ 导出类在 lib/presentation/ 的消费关系。"""
import ast, pathlib, sys

ROOT = pathlib.Path(__file__).resolve().parents[1]
PLATFORM = ROOT / "lib" / "platform"
PRESENTATION = ROOT / "lib" / "presentation"

# 守卫清单：类名 →（所在平台目录, 说明）
GUARDS = {
    "LongPressSpeedController": ("player", "长按倍速"),
    "SubtitleParser": ("player", "SRT/VTT/ASS 字幕解析"),
    "AutoPlayService": ("player", "自动连播"),
    "RemuxProxyServer": ("player", "转封装代理 18081"),
    "GoProxyClient": ("runtime", "Go 代理 10078"),
    "PanPlayer": ("player", "网盘播放编排"),
    "DirectLinkValidator": ("player", "直链校验"),
    "BackgroundPlayController": ("player", "后台播放"),
    "PipController": ("player", "画中画"),
}

def exported_classes():
    """提取 lib/platform/ 下所有顶层类名 → 文件路径。"""
    out = {}
    for f in PLATFORM.rglob("*.dart"):
        try:
            tree = ast.parse(f.read_text(encoding="utf-8"))
        except SyntaxError:
            continue
        for node in ast.walk(tree):
            if isinstance(node, ast.ClassDeclaration):
                out[node.name] = f
    return out

def presentation_imports():
    """收集 presentation 层全部 import 语句 + 标识符引用。"""
    texts = []
    for f in PRESENTATION.rglob("*.dart"):
        texts.append(f.read_text(encoding="utf-8"))
    return "\n".join(texts)

def main():
    classes, corpus, failures = exported_classes(), presentation_imports(), []
    for name, (_, desc) in GUARDS.items():
        if name not in classes:
            failures.append(f"[MISSING] {name}（{desc}）平台层不存在")
        elif name not in corpus:
            failures.append(f"[UNWIRED] {name}（{desc}）presentation 零消费")
    if failures:
        print("\n".join(failures)); sys.exit(1)
    print("platform wiring: OK")

if __name__ == "__main__":
    main()
```

**接入 CI**：`.github/workflows/flutter-check.yml` 追加一步 `- run: python3 scripts/check_platform_wiring.py`

  Future<void> play();

  Stream<Duration> get positionStream;

    r.fileName.isNotEmpty ? '${r.driveType}:${r.fileName}:${r.fileSize}'
                          : '${r.driveType}:${r.playID}'; // 缓存键不再用易变 playToken

// F-P27：vboxParams fragment
// 详情页剧集 URL 拼 #vbox_fid=xx&vbox_fsid=xx&vbox_node=xx；播放侧 Uri.fragmentParameters 解析定位

// F-P28：内核策略
KernelAdapter kernelForDrive(CloudDriveType t) => switch (t) {
  CloudDriveType.baidu => MediaKitAdapter(profile: 'mpv-safe'),  // 大文件 HEVC
  CloudDriveType.quark => MediaKitAdapter(profile: 'hdr'),        // 4K HDR
  CloudDriveType.uc    => NativeBridgeAdapter('ijk'),            // CDN Range
  _ => MediaKitAdapter.defaultProfile,
};
